Configuration Reference

Detailed documentation of every key and option in config.yml.

This is the default config.yml generated inside plugins/FindTheBlock/ with detailed explanations for each property.

plugins/FindTheBlock/config.yml 
# =====================================================================
#             FindTheBlock Configuration
# =====================================================================

lobby:
  world-name: "lobby"
  # Automatic void world & 100x100 bedrock platform generation
  platform:
    auto-generate: true
    size: 100
    y-level: 100
    material: "BEDROCK"
  spawn:
    x: 0.5
    y: 101.0
    z: 0.5
    yaw: 0.0
    pitch: 0.0
  # Lobby Protections
  protect-blocks: true
  disable-hunger: true
  disable-pvp: true
  disable-fall-damage: true
  disable-mob-spawning: true
  disable-item-drops: true
  disable-weather: true
  daylight-fixed: true
  # Teleport players back to world spawn before taking void damage
  void-teleport: true
  # Blocks above the void damage boundary (minHeight - 64) where teleport triggers
  void-teleport-blocks-above-damage: 2.0
  # If a player slips off the platform, teleport them back to spawn at this Y level


# =====================================================================
#                 Background World Factory Pipeline
# =====================================================================
world-factory:
  # Standby warm world pairs (Overworld + Nether) kept ready in queue
  buffer-size: 1
  # Radius in chunks around spawn to pre-generate asynchronously (6 = 13x13 chunks)
  chunk-radius: 6
  # Max chunks generated per second to guarantee ZERO server tick drops
  chunks-per-second: 30
  # Maximum world border
  match-border-size: 59999968.0

# =====================================================================
#                          Match Settings
# =====================================================================
match:
  # Maximum podium winners (1st, 2nd, 3rd) scaling dynamically with player count
  max-winners: 3
  # Duration in seconds of victory celebration before returning to lobby
  celebration-seconds: 10
  # 20-minute survival reward: Speed II & Regeneration I (Infinite)
  twenty-minute-buffs: true
  # Match timeout in hours before automatic cancellation
  timeout-hours: 3

  starter-kit:
    iron-tools: true
    silk-touch-pickaxe: true

  # Mid-game connection handling
  mid-game:
    allow-late-entry: true
    allow-spectate: true
    reconnect-grace-seconds: 60

  roulette:
    roll-seconds: 3
    preview-seconds: 4

  party:
    leader-inactivity-minutes: 10
